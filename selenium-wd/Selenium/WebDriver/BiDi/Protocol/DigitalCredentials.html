def set_virtual_wallet_behavior(
  action:,
  context: Serialization::UNSET,
  protocol: Serialization::UNSET,
  response: Serialization::UNSET
)
  Serialization.validate!('action', action, DigitalCredentials::VIRTUAL_WALLET_ACTION)
  params = SetVirtualWalletBehaviorParameters.new(
    action: action,
    context: context,
    protocol: protocol,
    response: response
  )
  execute(cmd: 'digitalCredentials.setVirtualWalletBehavior', params: params)
end

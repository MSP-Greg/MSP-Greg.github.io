      Overview

      A memoized accessor per generated domain, for a bridge to include. The includer supplies #connection; the domain classes load on first use, so this file loads without them.

def bidi_domain(name)
  connection = self.connection
  raise Error::WebDriverError, connection_error_message unless connection

  require 'selenium/webdriver/bidi/protocol'
  Protocol.const_get(name, false).new(connection)
end

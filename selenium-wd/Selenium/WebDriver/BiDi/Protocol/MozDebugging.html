def get_script_source(context:, script_url:)
  params = GetScriptSourceParameters.new(context: context, script_url: script_url)
  execute(cmd: 'moz:debugging.getScriptSource', params: params, result: MozDebugging::GetScriptSourceResult)

    user_contexts: user_contexts

def initialize(connections, uri, proxy_uri, size)
  @connections = connections
  @uri = uri
  @proxy_uri = proxy_uri
  @cert_files = connections.cert_files
  @queue = Thread::SizedQueue.new(size)
  size.times { @queue.push(nil) }
end

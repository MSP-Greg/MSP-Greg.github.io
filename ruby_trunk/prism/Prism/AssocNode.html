def ===(other)
  other.is_a?(AssocNode) &&
    (key === other.key) &&
    (value === other.value) &&
    (operator_loc.nil? == other.operator_loc.nil?)
end

def operator_loc
  location = @operator_loc
  case location
  when nil
    nil
  when Location
    location
  else
    @operator_loc = Location.new(source, location >> 32, location & 0xFFFFFFFF)
  end
end

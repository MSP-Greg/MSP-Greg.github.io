def map_type_name
  mapped_args = args.map { |type| type.map_type_name { yield _1 } }

  self.class.new(
    location:,
    prefix_location:,
    keyword_location:,
    colon_location:,
    name: yield(name),
    args: mapped_args,
    open_bracket_location:,
    close_bracket_location:,
    args_comma_locations:,
    comment_location:
  ) #: self
end

def type_fingerprint
  [
    "annots/module_self",
    name.to_s,
    args.map(&:to_s),
    comment_location&.source
  ]
end

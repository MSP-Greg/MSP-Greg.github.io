def deconstruct_keys(keys) # :nodoc:
  (keys || %i[node_id location]).each_with_object(
    {} #: Hash[Symbol, untyped]
  ) do |key, deconstructed|
    case key
    when :node_id                 then deconstructed[:node_id] = self.node_id
    when :location                then deconstructed[:location] = self.location
    end
  end
end

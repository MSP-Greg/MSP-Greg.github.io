def middlewares=(middlewares)
  stack = middleware_stack.dup
  stack.middlewares = middlewares
  replace_middleware_stack(stack)
  middlewares
end

  @controller.middleware_stack = ActiveSupport::Ractors.try_make_shareable(stack)
end

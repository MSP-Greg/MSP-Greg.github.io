def _value_of(record, normalizers = nil)
  value = record._read_attribute(@name)
  normalizer = normalizers&.first
  normalizer ? normalizer.call(value) : value
end

def resolve_aliases(owner)
  aliaz = owner.resolve_alias(@name)

  if aliaz == @name
    self
  else
    Key.for(aliaz)
  end
end

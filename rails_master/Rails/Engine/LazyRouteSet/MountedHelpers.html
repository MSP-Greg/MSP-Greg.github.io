      Overview

      Mounted helpers are only defined when mount runs during the route draw. _path/_url names are left to method_missing_module, which owns them and relies on reload_routes_unless_loaded returning true only to the caller that performed the load.

def method_missing(method_name, ...)
  if method_name.end_with?("_path", "_url")
    super
  else
    Rails.application&.reload_routes_unless_loaded
    if ActionDispatch::Routing::RouteSet::MountedHelpers.method_defined?(method_name)
      public_send(method_name, ...)
    else
      super
    end
  end
end

def respond_to_missing?(method_name, include_private = false)
  if method_name.end_with?("_path", "_url")
    super
  else
    Rails.application&.reload_routes_unless_loaded
    ActionDispatch::Routing::RouteSet::MountedHelpers.method_defined?(method_name) || super
  end
end

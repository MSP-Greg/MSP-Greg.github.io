      Overview

      Records marshaled in the ::Rails 6.1 format have their instance variables dumped directly, so their attribute sets load as this class (the name of the lazy attribute set at the time), with their attributes in Hashes by name. The first use builds every attribute and then works by name, like an AttributeSet. Its layout isn't the model's, so it can't be read by slot: the record lays it out like the model's first (see ActiveRecord::AttributeMethods::Read#_read_attribute_at!).

def build_attribute(name)
  type = @additional_types.fetch(name) { @types[name] } || Type.default_value

  if @values.key?(name)
    Attribute.from_database(name, @values[name], type, @casted_values.fetch(name) { nil })
  else
    default_keys = @default_keys if defined?(@default_keys)
    if (default_keys.nil? || default_keys.include?(name)) && (attr = @default_attributes.fetch(name) { nil })
      attr.dup
    else
      Attribute.uninitialized(name, type)
    end
  end
end

def build_attributes
  built = @attributes
  names = @values.keys | @types.keys | built.keys
  attributes = names.index_with { |name| built[name] || build_attribute(name) }
  @attributes = IndexedRow::Mutable[attributes]
end
